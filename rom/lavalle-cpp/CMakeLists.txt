cmake_minimum_required(VERSION 3.16)
project(lavalle_thinfilm LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(HDF5 REQUIRED COMPONENTS CXX)
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_executable(thinfilm ${SOURCES})
target_include_directories(thinfilm PRIVATE include ${HDF5_INCLUDE_DIRS})
target_link_libraries(thinfilm PRIVATE ${HDF5_CXX_LIBRARIES} ${HDF5_LIBRARIES})
target_compile_options(thinfilm PRIVATE -Wall -Wextra -Wpedantic)
